      set last_error_message=coalesce(last_error_message,'')
        ||' | scheduler_cleanup='||coalesce(v_unschedule_error,'UNKNOWN'),
          updated_at=clock_timestamp()
      where control_id=1;
    end;

    return jsonb_build_object(
      'status','FAILED',
      'sqlstate',v_sqlstate,
      'error',v_message
    );
  end;
